# The FFmpeg components the minimal build enables after --disable-everything (DECISIONS.md D046 §13). One "kind name"
# per line. build.sh turns them into --enable-<kind>=<name>; Test-MinimalFfmpeg.ps1 checks the binaries against them.
# Anything configure selects on its own for these (e.g. h263 for mpeg4, vp8 for webp, vp9_superframe_split, the
# filter graph's buffers) is allowed in the result; anything else is reported.

decoder pcm_s32le
decoder pcm_f32le

# Encoders — the export (libx264, aac) and the decoders' outputs to the application (rawvideo BGRA, f32le); nothing else.

encoder rawvideo
